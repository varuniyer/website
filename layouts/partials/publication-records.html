(slice .) . }}{{ end }} {{- $records := slice }} {{- range hugo.Data.publications }} {{- $row := . }} {{- if ne (printf "%T" .selected) "bool" }}{{ errorf "%s: selected must be true or false" .anthology_id }}{{ end }} {{- $mods := (resources.GetRemote (printf "https://aclanthology.org/%s.xml" .anthology_id) | transform.Unmarshal).mods }} {{- $modsNames := partial "inline/as-list.html" $mods.name }} {{- $last := sub (len $modsNames) 1 }} {{- range .equal_contributors }}{{ if or (lt . 0) (gt . $last) }}{{ errorf "%s: equal_contributors lists byline position %d, outside 0..%d" $row.anthology_id . $last }}{{ end }}{{ end }} {{- $authors := slice }} {{- range $i, $_ := $modsNames }} {{- $parts := dict }} {{- range partial "inline/as-list.html" .namePart }}{{ $parts = merge $parts (dict (index . "-type") (index . "#text")) }}{{ end }} {{- if not $parts.family }}{{ errorf "%s: author %d has no family name" $row.anthology_id $i }}{{ end }} {{- $name := strings.TrimSpace (printf "%s %s" (or $parts.given "") $parts.family) }} {{- $authors = $authors | append (dict "separator" (cond (eq $i 0) "" (cond (lt $i $last) ", " (cond (eq $last 1) " and " ", and "))) "name" $name "is_me" (eq $name site.Params.author) "equal_contribution" (in $row.equal_contributors $i)) }} {{- end }} {{- if not (where $authors "is_me" true) }}{{ errorf "%s: no author matches params.author" $row.anthology_id }}{{ end }} {{- $records = $records | append (dict "title" $mods.titleInfo.title "url" $mods.location.url "authors" $authors "venue" $mods.relatedItem.titleInfo.title "year" (index (split $mods.originInfo.dateIssued "-") 0) "selected" .selected "award" .award) }} {{- end }} {{- return $records }}
